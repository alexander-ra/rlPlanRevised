"""Standalone evaluation of the latest checkpoint, run as an isolated subprocess
by the daemon so a crash here can never take down training.

  python run_eval.py <run_dir>

Loads the newest valid checkpoint (slot_key + stratsum only), plays the blueprint
vs each baseline in duplicate-seat matches, and appends the bb/100 results to
<run_dir>/eval/index.json (+ a per-iteration file). Also dumps the 169-class
preflop raise frequency for the dashboard heatmap.
"""
from __future__ import annotations

import json
import sys
import time
from pathlib import Path

HERE = Path(__file__).resolve().parent
SRC = HERE.parent / "src"
sys.path.insert(0, str(SRC))


def main(run_dir):
    run_dir = Path(run_dir)
    import checkpoint as CK
    import evalmatch
    cfg = json.loads((HERE.parent / "config" / "default.json").read_text())
    n_decks = cfg["eval"].get("pilot_decks", 2000)

    ckpt, it = CK.latest_checkpoint(run_dir)
    if ckpt is None:
        print("no checkpoint yet; skipping eval")
        return
    slot_key, stratsum, meta = CK.load_strategy(ckpt)
    res = {"iteration": int(meta["iteration"])}
    for name, code in [("random", evalmatch.RANDOM),
                       ("calling_station", evalmatch.CALLING_STATION),
                       ("tag", evalmatch.TAG)]:
        t = time.time()
        bb, ci, _ = evalmatch.duplicate_match(slot_key, stratsum, code,
                                              n_decks=n_decks)
        res[name] = round(bb, 2)
        res[name + "_ci"] = round(ci, 2)
        print(f"  vs {name}: {bb:+.1f} bb/100 (+/-{ci:.0f}) [{time.time()-t:.0f}s]",
              flush=True)

    edir = run_dir / "eval"
    edir.mkdir(exist_ok=True)
    idx = edir / "index.json"
    hist = json.loads(idx.read_text()) if idx.exists() else []
    hist.append(res)
    tmp = str(idx) + ".tmp"
    Path(tmp).write_text(json.dumps(hist))
    import os
    os.replace(tmp, idx)
    (edir / f"eval_{res['iteration']}.json").write_text(json.dumps(res))
    print(f"eval @ {res['iteration']:,} written", flush=True)


if __name__ == "__main__":
    main(sys.argv[1])

Two genuinely deep questions. Let me take them in turn, because they're connected — and your dealing idea is the more powerful of the two.

1. What bucket size actually changes in the AI's "thinking"
A bucket is the unit of perception. On each street the AI sorts every possible (my cards + board) into N buckets by equity/equity-shape, and within a bucket it plays identically — it literally cannot tell those hands apart. So bucket count is the resolution of the AI's hand-reading.

Concretely, on the river with your two hole cards on a given board:

50 buckets ≈ 50 strength tiers. The nut flush and the second-nut flush likely fall in the same bucket → the AI value-bets and bluff-catches them the same way. It thinks in broad strokes: "strong / medium / weak," ~50 shades. Its blind spot: it can't tell "the nuts" from "very strong," so it can't extract the last bit of value or make the thin fold.
200 buckets ≈ 200 tiers. Now it separates the nuts from near-nuts, distinguishes a marginal bluff-catcher from a clear call. Finer value-betting, better bluff-catching in principle.
400 buckets — finer still.
But here's the catch, and it's the whole story: buckets set the ceiling of nuance; training determines whether that ceiling is reached. More buckets = 4× fewer training visits per bucket. A fine bucket that's barely trained plays worse than a coarse bucket that's well-trained, because regret-matching on a handful of visits produces erratic, high-variance strategies. It's the difference between a small vocabulary used correctly and a big vocabulary used wrong — the fine distinctions exist but aren't reliably applied. That's exactly what you're seeing now: 200 postflop buckets, but only ~13% of them meaningfully trained, so postflop it's playing its big vocabulary badly.

For your specific plan — a base for a self-play NN — this argues toward coarser. The NN's whole job is to grow the fine distinctions through self-play. What it needs from the base is stability and correctness in the broad strokes, not a high-resolution-but-erratic teacher. A well-converged 50–100 bucket base is a cleaner foundation than an under-converged 200. The NN adds the resolution; the base should add the reliability.

2. Your dealing idea — this is the better lever, and it's real
Your intuition is exactly right and it's a known technique (targeted / subgame-focused sampling). The inefficiency you're sensing: most hands end preflop. Roughly 60–80% of deals fold out before a flop, so the majority of iterations only touch already-converged preflop nodes and never train the postflop table at all. That's why postflop coverage is crawling — you're spending most of your 146k iterations/sec re-confirming that 72o folds.

Two ways to fix it, and the difference between them is important:

(a) Guided dealing (bias the deal toward hands that continue). Use the converged preflop strategy to avoid "wasting" deals: if a dealt hand would just fold preflop, replace it with one from the continuing range. More hands reach the flop → more postflop training per second. The critical caveat: CFR's math assumes chance is sampled from the true (uniform) distribution. If you bias the deal, you must multiply the counterfactual values by (true_prob / biased_prob) — an importance-sampling correction — or the learned strategy is biased toward the lines you over-dealt. It's doable, but the correction adds variance.

(b) Flop-start curriculum (the stronger version, and closest to what you described). Instead of always starting at preflop, for a fraction of iterations start at the flop: use the preflop strategy to sample a realistic preflop line (this fixes the pot size, who's in, and their positions), give each continuing player a hole-card range consistent with their preflop action, deal a flop, and train from there. This is essentially subgame solving folded into training — and it's powerful because your converged preflop strategy already defines the correct distribution of postflop situations (which hands, what pots, what ranges). You're not inventing scenarios; you're sampling the ones the game actually produces, just skipping the preflop replay. Because the ranges entering the flop are self-consistent with the preflop strategy, the postflop regrets converge to the right values — no bias, unlike naive guided dealing.

The beautiful part: this changes the bucket calculus. The reason I leaned coarser in Part 1 was convergence per bucket. If a flop-start curriculum gives you, say, 5–10× more postflop training visits, then 200 (or even 400) buckets get enough coverage to actually converge — you'd get the fine resolution and the stability. The dealing idea and the bucket count aren't separate decisions; better postflop sampling is what makes finer buckets affordable.

Your instinct is a known-good one — what you're describing has a name and a clean mathematical form, and your two ideas (retire preflop, flatten the dealing) actually compose into one coherent design. Let me think through it properly, including where the correctness landmines are.

1. Your flattening idea, formalized
What you described is tempered sampling: instead of sampling situations with their true probability p, sample with q ∝ p^α for some α < 1. At α=1 you have today's unbiased dealing; at α=0 everything is uniform. Your numeric example (5000:1 ratio compressed to 3.5:1) corresponds to a very aggressive α ≈ 0.15. For calibration:

α	5000:1 becomes
1.0 (today)	5000:1
0.5	71:1
0.25	8.4:1
0.15 (your example)	3.6:1
The deep question is why this is justified, and it's worth saying precisely: uniform dealing optimizes EV-weighted convergence — every spot gets exactly as much training as it contributes to expected value. A 1-in-a-million line gets a millionth of the attention because it's a millionth of your EV. That's the right allocation against the self-play distribution. But at deployment, opponents choose actions adversarially — they can steer the game into your rarest lines on purpose. Rare-in-self-play ≠ rare-in-real-play. So flattening reallocates training from "maximize EV against equilibrium opponents" toward "never be embarrassingly bad anywhere" — worst-case competence over average-case polish. For your stated purpose — a stable base that an NN later refines — uniform competence is arguably the more correct objective than EV-optimality. So: not a hack, a legitimate objective change.

The catch is bias. CFR's guarantees assume chance is sampled truly. Biased sampling needs an importance correction (multiply sampled utilities by p/q), and here's the pleasant surprise when you check the numbers: oversampled rare spots get weights below 1 (harmless), and your undersampled common spots only get weights like 0.5/0.35 ≈ 1.4 (mild variance). So even aggressive flattening is numerically tame where the correction is applied correctly. The real cost isn't variance — it's that you're deliberately spending, say, 10% of compute on spots worth 0.01% of EV. That's the trade you're choosing, eyes open.

2. Your "retire preflop" idea — right direction, one trap
The trap: the current preflop converged against near-random postflop play. Some preflop calls look profitable only because opponents play the later streets badly. As postflop tightens up, the correct preflop strategy shifts (fewer loose calls, different 3-bet mix). Hard-freezing preflop now would lock in those distortions permanently — the base would have a subtly wrong trunk that the NN inherits.

The fix is a soft freeze: keep ~20–25% of iterations as normal full-game training (preflop keeps adapting slowly as postflop improves, and the average-strategy weighting stays honest), and spend the other ~75–80% on postflop-focused iterations. You get almost all the reallocation benefit without severing the feedback loop between streets. Later (say, 10× more iterations in), if preflop has stopped moving, a hard freeze becomes safe.

3. How the pieces compose — three iteration types
This is the design I'd propose, in increasing order of statistical risk:

Type A — normal full-game (~25%). Exactly today's iteration. Keeps the trunk adapting and the math grounded.

Type B — flop-start via rejection (~60%). Play preflop quickly with the current strategy, all seats just sampling (no branching, no updates). If the hand ends preflop, throw it away and redeal — repeat until a flop is reached, then run the normal learning traversal from there. The elegance of rejection sampling: the flop situations you train on are distributed exactly as the real game produces them — correct ranges, correct pot sizes, correct multiway frequencies — with zero importance weights and zero bias (for the postflop game given the current trunk). Preflop sampling costs microseconds, so "wasting" folded deals is nearly free. This alone removes the "most iterations die preflop" waste completely.

Type C — tempered/flattened starts (~15%). This is your flattening, applied where it's safest: bias the traverser's own hole cards toward uniform-over-169-classes (rather than natural frequencies), with the exact p/q correction on its utilities — mathematically clean, low variance, and it directly attacks "my rare hand types never get trained." Optionally, also nudge preflop lines toward under-visited betting sequences (the 3-bet/4-bet pots that rarely occur in self-play) with tempering on the line choice. I'd start α ≈ 0.5 rather than your 0.15 — flatten by 70×, see the coverage response, and turn the dial from evidence rather than committing to near-uniform on day one.

One more subtle benefit of B+C worth naming: opponents who continue slightly more often than equilibrium (which tempering induces at the margin) resemble human populations more than Nash does — humans are loose. For a base that will later be exploited-against and compared to real data, that's a feature, not a bug, as long as we keep it mild and the Type-A iterations anchor the trunk.

@njit(cache=True)
def _playout(rules, sk, regret, kpre, bpre, kflop, bflop, kturn, bturn,
             kriver, briver, binom, perms, end_street, dec_street, seen_flop):
    """One full hand, all seats sampling regret-matched current strategy."""
    deck = np.arange(52)
    for i in range(51, 0, -1):
        j = np.random.randint(0, i + 1)
        t = deck[i]
        deck[i] = deck[j]
        deck[j] = t
    holes = deck[:12].astype(np.int32)
    board = deck[12:17].astype(np.int32)
    st = EN.new_hand_nb(rules, holes, board, 0)
    ok = np.empty(8, dtype=np.int64)
    oa = np.empty(8, dtype=np.int64)
    strat = np.empty(8, dtype=np.float64)
    got_flop = 0
    while st[EN.S_FIN] == 0:
        p = st[EN.S_TOACT]
        s = st[EN.S_STREET]
        if s >= 1:
            got_flop = 1
        dec_street[s] += 1
        bucket = mccfr.player_bucket(st, p, kpre, bpre, kflop, bflop,
                                     kturn, bturn, kriver, briver, binom, perms)
        h = infoset_hash(st, bucket)
        slot = mccfr.table_find(sk, h)
        n = EN.legal_actions_nb(st, rules, ok, oa)
        if slot >= 0:
            mccfr.regret_matching(regret[slot], n, strat)
        else:
            for a in range(n):
                strat[a] = 1.0 / n
        r = np.random.random()
        c = 0.0
        a_pick = n - 1
        for a in range(n):
            c += strat[a]
            if r < c:
                a_pick = a
                break
        EN.apply_action_nb(st, rules, ok[a_pick], oa[a_pick])
    end_street[st[EN.S_STREET]] += 1
    seen_flop[0] += got_flop


@njit(cache=True)
def playout_study(n_hands, rules, sk, regret, kpre, bpre, kflop, bflop,
                  kturn, bturn, kriver, briver, binom, perms):
    end_street = np.zeros(4, dtype=np.int64)
    dec_street = np.zeros(4, dtype=np.int64)
    seen_flop = np.zeros(1, dtype=np.int64)
    for _ in range(n_hands):
        _playout(rules, sk, regret, kpre, bpre, kflop, bflop, kturn, bturn,
                 kriver, briver, binom, perms, end_street, dec_street, seen_flop)
    return end_street, dec_street, seen_flop[0]


@njit(parallel=True, cache=True)
def coverage_scan(bases, streets, bps, sk, regret, stratsum, n_chunks):
    """Per-chunk per-street counters:
    [total, regret_nz, strat_nz, strat>1, strat>10, strat>100, strat_sum]"""
    out = np.zeros((n_chunks, 4, 7), dtype=np.float64)
    nb_ = bases.shape[0]
    chunk = (nb_ + n_chunks - 1) // n_chunks
    for c in prange(n_chunks):
        lo = c * chunk
        hi = min(lo + chunk, nb_)
        for i in range(lo, hi):
            s = streets[i]
            base = bases[i]
            nbk = bps[s]
            for b in range(nbk):
                h = mix_bucket(base, b)
                slot = mccfr.table_find(sk, h)
                out[c, s, 0] += 1
                if slot < 0:
                    continue
                rnz = False
                for a in range(4):
                    if regret[slot, a] != 0.0:
                        rnz = True
                        break
                if rnz:
                    out[c, s, 1] += 1
                ssum = 0.0
                for a in range(4):
                    ssum += stratsum[slot, a]
                if ssum > 0.0:
                    out[c, s, 2] += 1
                if ssum > 1.0:
                    out[c, s, 3] += 1
                if ssum > 10.0:
                    out[c, s, 4] += 1
                if ssum > 100.0:
                    out[c, s, 5] += 1
                out[c, s, 6] += ssum
    return out
