% ---------------------------------------------------------------------------
% scripts/float_layout.tex  —  included via --include-in-header
%
% These documents are figure-dense: several chapters put four or more figures in
% a row with little text between them. LaTeX's float mechanism does not cope,
% and its failures here are SILENT — the content is in the generated .tex, is
% absent from the PDF, and no error is raised:
%
%   * a section heading sitting between two floats is not typeset
%     (chapter 11's "Coalition-aware MAPPO" disappeared this way)
%   * a figure's \caption is dropped while its image is kept
%     (chapter 10's Mini-PBT figure; it left a hole in the volume-wide figure
%     numbering, 60 -> 62, which is how it was finally noticed)
%
% Two earlier attempts to tune the float parameters each fixed one symptom and
% caused the other: a \FloatBarrier at every \subsection protected the headings
% but killed the caption of any figure ending a subsection right after a table
% (pandoc renders every markdown table as a longtable); dropping back to
% placeins' [section] barrier restored those captions and lost the headings
% again, in the English chapters where the shorter text changes the pagination.
%
% So the floats are simply not floated. \floatplacement{...}{H} pins every
% figure and table exactly where the author put it: no queue to overflow, no
% barrier interaction, nothing to drift or be dropped, and a figure always sits
% with the text that introduces it. The cost is looser page breaks, which for a
% report-style document reads fine and is the right trade for correctness.
%
% scripts/check_captions.py and scripts/check_headings.py assert both symptoms
% stay fixed; run them after any change here.
% ---------------------------------------------------------------------------

\usepackage{float}

\floatplacement{figure}{H}
\floatplacement{table}{H}
